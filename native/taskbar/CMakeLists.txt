cmake_minimum_required(VERSION 3.24)
project(MagmPaperTaskbar LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The taskbar component requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /EHsc /permissive- /utf-8 /guard:cf)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(MagmTaskbar SHARED plugin.cpp styles.cpp exports.def)
target_link_libraries(MagmTaskbar PRIVATE windowsapp user32 comctl32 ole32 shell32)
add_executable(MagmTaskbarHost host.cpp app.manifest)
target_link_libraries(MagmTaskbarHost PRIVATE windowsapp user32 shell32 ole32)
add_executable(MagmTaskbarTests tests.cpp styles.cpp app.manifest)
target_link_libraries(MagmTaskbarTests PRIVATE windowsapp user32 ole32)
target_compile_definitions(MagmTaskbarTests PRIVATE MAGM_HARNESS)
add_library(MagmTaskbarHarness SHARED plugin.cpp styles.cpp exports.def)
target_compile_definitions(MagmTaskbarHarness PRIVATE MAGM_HARNESS)
target_link_libraries(MagmTaskbarHarness PRIVATE windowsapp user32 comctl32 ole32 shell32)
add_executable(MagmTaskbarTestHost host.cpp app.manifest)
target_compile_definitions(MagmTaskbarTestHost PRIVATE MAGM_HARNESS)
target_link_libraries(MagmTaskbarTestHost PRIVATE windowsapp user32 shell32 ole32)
add_dependencies(MagmTaskbarTests MagmTaskbarHarness MagmTaskbarTestHost)
enable_testing()
add_test(NAME native-taskbar COMMAND MagmTaskbarTests)
set_tests_properties(native-taskbar PROPERTIES TIMEOUT 90)
